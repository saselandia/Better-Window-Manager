cmake_minimum_required(VERSION 3.25)
project(bwm-plasma-desktops LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND_SERVER REQUIRED IMPORTED_TARGET wayland-server>=1.22)
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
find_file(PLASMA_DESKTOP_XML org-kde-plasma-virtual-desktop.xml PATH_SUFFIXES plasma-wayland-protocols PATHS /usr/share /usr/local/share REQUIRED)
# KDE marks this protocol private and permits incompatible changes without a bump.
file(SHA256 ${PLASMA_DESKTOP_XML} BWM_DESKTOP_XML_HASH)
if(NOT BWM_DESKTOP_XML_HASH STREQUAL "8551063295644bceaaaca2757aa5f99f3855c6ef50c7200eb74462faab9f3fa9")
    message(FATAL_ERROR "Plasma desktop protocol XML changed; review events/requests and native tests before updating the pin")
endif()
add_custom_command(OUTPUT plasma-desktops-server.h COMMAND ${WAYLAND_SCANNER} server-header ${PLASMA_DESKTOP_XML} plasma-desktops-server.h DEPENDS ${PLASMA_DESKTOP_XML})
add_custom_command(OUTPUT plasma-desktops-protocol.c COMMAND ${WAYLAND_SCANNER} private-code ${PLASMA_DESKTOP_XML} plasma-desktops-protocol.c DEPENDS ${PLASMA_DESKTOP_XML})
set(BWM_DESKTOP_CONTROL_XML ${CMAKE_CURRENT_SOURCE_DIR}/bwm-plasma-desktop-control-v1.xml)
add_custom_command(OUTPUT desktop-control-server.h COMMAND ${WAYLAND_SCANNER} server-header ${BWM_DESKTOP_CONTROL_XML} desktop-control-server.h DEPENDS ${BWM_DESKTOP_CONTROL_XML})
add_custom_command(OUTPUT desktop-control-protocol.c COMMAND ${WAYLAND_SCANNER} private-code ${BWM_DESKTOP_CONTROL_XML} desktop-control-protocol.c DEPENDS ${BWM_DESKTOP_CONTROL_XML})
add_library(bwm-desktop-protocol STATIC desktops.cpp provider.cpp countmirror.cpp activation.cpp workspaces.cpp ${CMAKE_CURRENT_BINARY_DIR}/plasma-desktops-protocol.c ${CMAKE_CURRENT_BINARY_DIR}/plasma-desktops-server.h ${CMAKE_CURRENT_BINARY_DIR}/desktop-control-server.h ${CMAKE_CURRENT_BINARY_DIR}/desktop-control-protocol.c)
target_include_directories(bwm-desktop-protocol PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(bwm-desktop-protocol PUBLIC PkgConfig::WAYLAND_SERVER)
target_compile_options(bwm-desktop-protocol PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fno-gnu-unique>)
add_executable(test-workspace-projection test-workspaces.cpp)
target_link_libraries(test-workspace-projection PRIVATE bwm-desktop-protocol)
pkg_check_modules(WAYLAND_CLIENT REQUIRED IMPORTED_TARGET wayland-client)
target_link_libraries(bwm-desktop-protocol PUBLIC PkgConfig::WAYLAND_CLIENT)
add_custom_command(OUTPUT desktop-control-client.h COMMAND ${WAYLAND_SCANNER} client-header ${BWM_DESKTOP_CONTROL_XML} desktop-control-client.h DEPENDS ${BWM_DESKTOP_CONTROL_XML})
add_custom_command(OUTPUT plasma-desktops-client.h COMMAND ${WAYLAND_SCANNER} client-header ${PLASMA_DESKTOP_XML} plasma-desktops-client.h DEPENDS ${PLASMA_DESKTOP_XML})
add_executable(test-desktop-provider test-provider.cpp ${CMAKE_CURRENT_BINARY_DIR}/desktop-control-client.h ${CMAKE_CURRENT_BINARY_DIR}/plasma-desktops-client.h)
target_include_directories(test-desktop-provider PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(test-desktop-provider PRIVATE bwm-desktop-protocol)
add_library(desktop-provider-teardown-module MODULE test-teardown-module.cpp)
target_link_libraries(desktop-provider-teardown-module PRIVATE bwm-desktop-protocol)
target_compile_options(desktop-provider-teardown-module PRIVATE -fno-gnu-unique)
# Keep the tested core's callbacks inside this DSO rather than interposing the
# executable's static copy. The test must really unmap their defining module.
target_link_options(desktop-provider-teardown-module PRIVATE -Wl,--exclude-libs,ALL)
target_link_libraries(test-desktop-provider PRIVATE ${CMAKE_DL_LIBS})
target_compile_definitions(test-desktop-provider PRIVATE BWM_TEST_PROVIDER_MODULE="$<TARGET_FILE:desktop-provider-teardown-module>")
add_dependencies(test-desktop-provider desktop-provider-teardown-module)
add_executable(test-desktop-control-wire test-control-wire.cpp ${CMAKE_CURRENT_BINARY_DIR}/desktop-control-client.h ${CMAKE_CURRENT_BINARY_DIR}/plasma-desktops-client.h)
target_include_directories(test-desktop-control-wire PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(test-desktop-control-wire PRIVATE bwm-desktop-protocol PkgConfig::WAYLAND_CLIENT)
add_executable(desktop-test-display test-display.cpp)
find_file(XDG_SHELL_XML xdg-shell.xml PATH_SUFFIXES wayland-protocols/stable/xdg-shell PATHS /usr/share /usr/local/share REQUIRED)
add_custom_command(OUTPUT test-xdg-shell-server.h COMMAND ${WAYLAND_SCANNER} server-header ${XDG_SHELL_XML} test-xdg-shell-server.h DEPENDS ${XDG_SHELL_XML})
add_custom_command(OUTPUT test-xdg-shell.c COMMAND ${WAYLAND_SCANNER} private-code ${XDG_SHELL_XML} test-xdg-shell.c DEPENDS ${XDG_SHELL_XML})
target_sources(desktop-test-display PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/test-xdg-shell-server.h ${CMAKE_CURRENT_BINARY_DIR}/test-xdg-shell.c)
target_include_directories(desktop-test-display PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(desktop-test-display PRIVATE bwm-desktop-protocol)
add_executable(desktop-provider-test-display test-display.cpp ${CMAKE_CURRENT_BINARY_DIR}/test-xdg-shell-server.h ${CMAKE_CURRENT_BINARY_DIR}/test-xdg-shell.c)
target_compile_definitions(desktop-provider-test-display PRIVATE BWM_ARBITRATED_PROVIDER=1)
target_include_directories(desktop-provider-test-display PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(desktop-provider-test-display PRIVATE bwm-desktop-protocol)
option(BWM_BUILD_HYPRLAND_PLUGIN "Build experimental compositor extension; never loads/installs it" OFF)
if(BWM_BUILD_HYPRLAND_PLUGIN)
    pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland pixman-1 libdrm libinput libudev xkbcommon lua5.4)
    find_package(nlohmann_json REQUIRED)
    add_library(bwm-plasma-desktops MODULE plugin.cpp)
    set_target_properties(bwm-plasma-desktops PROPERTIES PREFIX "" CXX_STANDARD 23)
    target_compile_options(bwm-plasma-desktops PRIVATE -fno-gnu-unique)
    target_link_libraries(bwm-plasma-desktops PRIVATE bwm-desktop-protocol PkgConfig::HYPRLAND nlohmann_json::nlohmann_json)
endif()
