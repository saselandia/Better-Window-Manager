cmake_minimum_required(VERSION 3.25)
project(bwm-systemsettings VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(KF6KCMUtils REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6I18n REQUIRED)
find_package(KDecoration3 REQUIRED)
find_package(KF6Config REQUIRED)
add_library(bwm-nightlight OBJECT nightlightmodule.cpp)
set_target_properties(bwm-nightlight PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(bwm-nightlight PUBLIC Qt6::Widgets KF6::KCMUtils KF6::CoreAddons KF6::I18n)
add_library(kcm_bwm_nightlight MODULE plugin.cpp)
target_link_libraries(kcm_bwm_nightlight PRIVATE bwm-nightlight)
set_target_properties(kcm_bwm_nightlight PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/plasma/kcms/systemsettings")
# Atomic replacement also protects a KCM already mapped by an open System Settings process.
install(CODE [[
    set(bwm_kcm_destination "$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}/lib/qt6/plugins/plasma/kcms/systemsettings")
    file(MAKE_DIRECTORY "${bwm_kcm_destination}")
    file(COPY_FILE "$<TARGET_FILE:kcm_bwm_nightlight>" "${bwm_kcm_destination}/.kcm_bwm_nightlight.so.tmp")
    file(RENAME "${bwm_kcm_destination}/.kcm_bwm_nightlight.so.tmp" "${bwm_kcm_destination}/kcm_bwm_nightlight.so")
]])
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(test-nightlight test-nightlight.cpp)
    target_link_libraries(test-nightlight PRIVATE bwm-nightlight Qt6::Test)
    target_compile_definitions(test-nightlight PRIVATE BWM_TEST_PLUGIN_ROOT="${CMAKE_BINARY_DIR}/plugins")
    add_dependencies(test-nightlight kcm_bwm_nightlight)
    add_test(NAME native-kcm COMMAND test-nightlight)
    set_tests_properties(native-kcm PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()

# R3: native pages over shared managers. No duplicate settings service.
add_library(bwm-manager OBJECT managermodule.cpp)
set_target_properties(bwm-manager PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(bwm-manager PUBLIC Qt6::Widgets KF6::KCMUtils KF6::CoreAddons KF6::I18n)
foreach(category cursor screens workspaces apps startup shortcuts energy windows effects)
    add_library(kcm_bwm_${category} MODULE plugin_${category}.cpp)
    target_link_libraries(kcm_bwm_${category} PRIVATE bwm-manager)
    set_target_properties(kcm_bwm_${category} PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/plasma/kcms/systemsettings")
    install(CODE "
        set(dest \"\$ENV{DESTDIR}${CMAKE_INSTALL_PREFIX}/lib/qt6/plugins/plasma/kcms/systemsettings\")
        file(MAKE_DIRECTORY \"\${dest}\")
        file(COPY_FILE \"$<TARGET_FILE:kcm_bwm_${category}>\" \"\${dest}/.kcm_bwm_${category}.so.tmp\")
        file(RENAME \"\${dest}/.kcm_bwm_${category}.so.tmp\" \"\${dest}/kcm_bwm_${category}.so\")
    ")
endforeach()
add_executable(bwm-kcm-catalog kcm-catalog.cpp)
target_link_libraries(bwm-kcm-catalog PRIVATE Qt6::Widgets KF6::KCMUtils KF6::CoreAddons KDecoration3::KDecoration)
install(TARGETS bwm-kcm-catalog RUNTIME DESTINATION bin)

if(BUILD_TESTING)
    add_executable(test-manager test-manager.cpp)
    target_link_libraries(test-manager PRIVATE bwm-manager Qt6::Test)
    target_compile_definitions(test-manager PRIVATE BWM_TEST_PLUGIN_ROOT="${CMAKE_BINARY_DIR}/plugins")
    add_test(NAME manager-kcm COMMAND test-manager)
    set_tests_properties(manager-kcm PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(test-decoration-render ../bwm-decorator/poc-fase0.cpp)
    target_link_libraries(test-decoration-render PRIVATE Qt6::Widgets KF6::CoreAddons KF6::ConfigCore KDecoration3::KDecoration KDecoration3::KDecoration3Private)
    add_executable(test-decoration-option test-decoration-option.cpp)
    target_link_libraries(test-decoration-option PRIVATE Qt6::Test Qt6::Widgets KF6::KCMUtils KF6::CoreAddons)
    add_test(NAME installed-decoration-option COMMAND test-decoration-option)
    set_tests_properties(installed-decoration-option PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
